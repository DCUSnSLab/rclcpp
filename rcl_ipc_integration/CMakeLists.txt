cmake_minimum_required(VERSION 3.16)
project(rcl_ipc_integration)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
pybind11_add_module(_ipc_native native.cpp)
target_compile_features(_ipc_native PRIVATE cxx_std_17)
target_link_libraries(_ipc_native PRIVATE rclcpp::rclcpp ${sensor_msgs_TARGETS})
install(TARGETS _ipc_native LIBRARY DESTINATION lib/${PROJECT_NAME})
ament_package()
